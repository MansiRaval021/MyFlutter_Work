import 'package:flutter/material.dart';

class RestaurantScreen extends StatefulWidget {
  const RestaurantScreen({super.key});

  @override
  State<RestaurantScreen> createState() => _RestaurantScreenState();
}

class _RestaurantScreenState extends State<RestaurantScreen> {
  List<Map<String, String>> restaurants = [
    {'name': 'Pizza Hut', 'cuisine': 'Italian'},
    {'name': 'Taj Palace', 'cuisine': 'Indian'},
    {'name': 'Sushi House', 'cuisine': 'Japanese'},
  ];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Saved Restaurants'),
      ),
      body: ListView.builder(
        itemCount: restaurants.length,
        itemBuilder: (context, index) {
          return Dismissible(
            key: Key(restaurants[index]['name']!),
            onDismissed: (_) {
              setState(() {
                restaurants.removeAt(index);
              });
            },
            background: Container(color: Colors.red),
            child: ListTile(
              title: Text(restaurants[index]['name']!),
              subtitle: Text(restaurants[index]['cuisine']!),
            ),
          );
        },
      ),
    );
  }
}
